/**
 * NOTESCSBS | Institutional Welcome Broadcast
 * -------------------------------------------
 * This script automates the delivery of a high-fidelity Welcome Brief 
 * when a student enrolls in the Academic Shift Registry.
 * 
 * Trigger: On Form Submit
 * Setup: Deploy as a trigger in Google Apps Script associated with your response sheet.
 */

function onFormSubmit(e) {
  // Capture captured email (adjust index if email is not in the first column after timestamp)
  const email = e.values[1]; 
  
  // Check for duplicates in the spreadsheet
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  
  let count = 0;
  // Assuming email is in the second column (index 1)
  for (let i = 0; i < data.length; i++) {
    if (data[i][1] === email) {
      count++;
    }
  }
  
  // If count is > 1, it's a duplicate submission (the current one makes it at least 1)
  if (count > 1) {
    // Delete the duplicate row to keep the spreadsheet clean
    sheet.deleteRow(e.range.getRow());
    // Stop execution so we don't send the email again
    return;
  }
  
  const subject = "Welcome to Notes CSBS - Registration Confirmed";
  
  // Clean, professional, brand-aligned email template to avoid spam filters
  const htmlBody = `
    <!DOCTYPE html>
    <html lang="en">
    <head>
      <meta charset="UTF-8">

      <style>
        body { 
          margin: 0; 
          padding: 0; 
          background-color: #f8fafc;

          -webkit-font-smoothing: antialiased;
        }
        .wrapper {
          padding: 40px 20px;
        }
        .container {
          max-width: 600px;
          margin: 0 auto;
          background-color: #ffffff;
          border-radius: 16px;
          border: 1px solid #e2e8f0;
          overflow: hidden;
          box-shadow: 0 4px 6px -1px rgba(0, 0, 0, 0.05);
        }
        .header {
          padding: 40px 40px 20px 40px;
          text-align: center;
          border-bottom: 1px solid #f8fafc;
        }
        .logo {
          font-weight: 900;

        }
        .logo-accent {

        }

          color: #475569;
        }
        .cta-container {
          text-align: center;
          margin-top: 40px;
          margin-bottom: 10px;
        }
        .button {
          display: inline-block;
          background-color: #2563eb;

        }
        .footer-text {
          font-size: 13px;

          margin: 0;
        }
      </style>
    </head>
    <body>
      <div class="wrapper">
        <div class="container">
          <div class="header">
            <div class="logo">NOTES<span class="logo-accent">CSBS</span></div>

            <h1>Registration Confirmed</h1>
            <p>
              Hello there,

            <p>
              Welcome to the Notes CSBS institutional registry. Your email has been successfully registered to receive real-time curriculum notifications and curated academic resource drops for BMSCE students.
            </p>

              We are dedicated to providing you with the highest quality lecture notes, previous year question papers (PYQs), and study materials across all semesters.

            <p class="footer-text">© Notes CSBS. All rights reserved.</p>
          </div>
        </div>
      </div>
    </body>
    </html>
  `;
  
  MailApp.sendEmail({
    to: email,
    subject: subject,
    htmlBody: htmlBody
  });
}
